380:       if (!msgId) throw new Error('解锁命令没发出去（没有 msgId）');
381:       return pollMsg(msgId, onProgress).then(res => {
382:         return { ok: true, msgId: msgId, tries: res.tries, detail: res.data };
383:       });
384:     });

387:   function pollMsg(msgId, onProgress) {
388:     let tries = 0;
389:     const MAX = 15;
390:     return new Promise((resolve, reject) => {
391:       const tick = () => {
392:         tries++;
393:         call('machine/msg/sync', { msgId: msgId }).then(d => {
394:           if (onProgress) onProgress(tries, (d && (d.state || d.code)) || '');
395:           if (!d) {                       // 还没回话
396:             if (tries >= MAX) { reject(new Error('机器一直没有响应（' + MAX + ' 次）')); return; }
397:             setTimeout(tick, 1000);
398:             return;
399:           }
400:           const state = String(d.state || '');
401:           const code = String(d.code || '');
402:           if (state === 'OK' || code === '05') { resolve({ tries: tries, data: d }); return; }
403:           const err = new Error('机器拒绝了这次解锁（' + (state || code || '未知') + '）');
404:           err.detail = d;
405:           reject(err);
406:         }).catch(e => {
407:           // 单次网络抖动不该判定失败，继续等到上限
408:           if (tries >= MAX) { reject(e); return; }
